# NOTA: los valores entre < > son las contraseñas y el secreto del laboratorio; no se publican en este repositorio.
# ==========================================================
# 05 - SW1-2241: AAA con RADIUS, SSH, logs y niveles de privilegio
# En la consola de SW1-2241 (GNS3)
# ==========================================================
enable
configure terminal

# ---------- Usuario local y SSH ----------
username admin-local privilege 15 secret <contraseña-local>
enable secret <contraseña-enable>
service password-encryption
ip domain-name aaa2241.local
crypto key generate rsa modulus 2048
ip ssh version 2
ip ssh time-out 60
ip ssh authentication-retries 3

# ---------- AAA y RADIUS ----------
aaa new-model

radius server NPS-2241
 address ipv4 10.22.41.10 auth-port 1812 acct-port 1813
 key <secreto-radius>
 exit

aaa group server radius GRP-NPS
 server name NPS-2241
 exit

aaa authentication login CON-AUX-AUTH group GRP-NPS local
aaa authorization console
aaa authorization exec CON-AUX-AUTHZ group GRP-NPS local
aaa accounting exec CON-AUX-ACCT start-stop group GRP-NPS
aaa accounting system default start-stop group GRP-NPS

aaa authentication login default local
aaa authorization exec default local

# ---------- Lineas: RADIUS en consola y auxiliar, SSH en vty ----------
line con 0
 login authentication CON-AUX-AUTH
 authorization exec CON-AUX-AUTHZ
 accounting exec CON-AUX-ACCT
 logging synchronous
 exec-timeout 10 0
 exit

line aux 0
 login authentication CON-AUX-AUTH
 authorization exec CON-AUX-AUTHZ
 accounting exec CON-AUX-ACCT
 exec-timeout 10 0
 exit

line vty 0 15
 transport input ssh
 exec-timeout 10 0
 length 0
 exit

# ---------- Logs ----------
service timestamps log datetime msec
service timestamps debug datetime msec
logging on
logging buffered 64000 informational
logging userinfo
login on-success log
login on-failure log

# ---------- Niveles de privilegio ----------
privilege exec level 10 show running-config
privilege exec level 10 show logging
privilege exec level 5 show ip interface brief
privilege exec level 5 ping
privilege exec level 5 show
end
write memory

# ---------- Verificacion y prueba de RADIUS ----------
show aaa servers
show ip ssh
test aaa group GRP-NPS user-nivel5 <contraseña-usuarios> new-code
